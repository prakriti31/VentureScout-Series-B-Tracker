        function populateDropdowns() {

            const industries = [...new Set(startupData.map(d => d.industry))].sort();

            const cities = [...new Set(startupData.map(d => d.city))].sort();

            

            const indSelect = document.getElementById('industryFilter');

            const citySelect = document.getElementById('cityFilter');

            

            industries.forEach(ind => {

                const opt = document.createElement('option');

                opt.value = ind;

                opt.textContent = ind;

                indSelect.appendChild(opt);

            });

            

            cities.forEach(city => {

                const opt = document.createElement('option');

                opt.value = city;

                opt.textContent = city;

                citySelect.appendChild(opt);

            });

        }


        function filterData() {

            state.data = startupData.filter(item => {

                const matchSearch = item.name.toLowerCase().includes(state.searchQuery) || 

                                    item.investor.toLowerCase().includes(state.searchQuery) ||

                                    item.desc.toLowerCase().includes(state.searchQuery);

                const matchInd = state.filterIndustry === 'All' || item.industry === state.filterIndustry;

                const matchCity = state.filterCity === 'All' || item.city === state.filterCity;

                return matchSearch && matchInd && matchCity;

            });

            

            // Re-sort after filter

            sortDataInternal();

            

            updateUI();

        }

        

        function sortTable(field) {

            // Toggle order if clicking same field

            if (state.sortField === field) {

                state.sortOrder = state.sortOrder === 'desc' ? 'asc' : 'desc';

            } else {

                state.sortField = field;

                state.sortOrder = 'desc'; // Default to high-to-low for new fields

            }

            

            sortDataInternal();

            updateUI();

        }

        

        function sortDataInternal() {

            state.data.sort((a, b) => {

                let valA = a[state.sortField];

                let valB = b[state.sortField];

                

                if (typeof valA === 'string') valA = valA.toLowerCase();

                if (typeof valB === 'string') valB = valB.toLowerCase();

                

                if (valA < valB) return state.sortOrder === 'asc' ? -1 : 1;

                if (valA > valB) return state.sortOrder === 'asc' ? 1 : -1;

                return 0;

            });

        }


        function updateUI() {

            renderTable();

            renderKPIs();

            renderCharts(); // Re-render charts with filtered data

        }

        function renderCharts() {

            // Prepare Data

            

            // 1. Industry (Top 5 + Others)

            const indCounts = {};

            state.data.forEach(d => {

                // Simplified industry grouping for chart

                let key = d.industry.split(' / ')[0]; 

                indCounts[key] = (indCounts[key] || 0) + d.amount; // Summing funding amount

            });

            const sortedInd = Object.entries(indCounts).sort((a,b) => b[1] - a[1]);

            const topInd = sortedInd.slice(0, 5);

            const otherIndVal = sortedInd.slice(5).reduce((acc, curr) => acc + curr[1], 0);

            if(otherIndVal > 0) topInd.push(['Others', otherIndVal]);


            const indLabels = topInd.map(i => i[0]);

            const indValues = topInd.map(i => i[1]);


            // 2. Location

            const locCounts = {};

            state.data.forEach(d => {

                locCounts[d.city] = (locCounts[d.city] || 0) + 1;

            });

            const sortedLoc = Object.entries(locCounts).sort((a,b) => b[1] - a[1]);

            const locLabels = sortedLoc.map(i => i[0]);

            const locValues = sortedLoc.map(i => i[1]);


            // 3. Deal Size Buckets

            const buckets = { '<$30M': 0, '$30-50M': 0, '$50-80M': 0, '$80M+': 0 };

            state.data.forEach(d => {

                if (d.amount < 30) buckets['<$30M']++;

                else if (d.amount < 50) buckets['$30-50M']++;

                else if (d.amount < 80) buckets['$50-80M']++;

                else buckets['$80M+']++;

            });

            const bucketLabels = Object.keys(buckets);

            const bucketValues = Object.values(buckets);


            // Chart Configuration Helpers

            const commonOptions = {

                responsive: true,

                maintainAspectRatio: false,

                plugins: {

                    legend: { position: 'bottom', labels: { boxWidth: 12, font: { size: 11 } } }

                }

            };


            // Render Chart 1: Industry (Bar)

            updateChart('industryChart', 'bar', {

                labels: indLabels,

                datasets: [{

                    label: 'Total Funding ($M)',

                    data: indValues,

                    backgroundColor: '#6366f1', // Indigo 500

                    borderRadius: 4

                }]

            }, {

                ...commonOptions,

                plugins: { legend: { display: false } }, // Hide legend for single bar

                scales: { y: { beginAtZero: true, grid: { color: '#f1f5f9' } }, x: { grid: { display: false } } }

            });


            // Render Chart 2: Location (Doughnut)

            updateChart('locationChart', 'doughnut', {

                labels: locLabels,

                datasets: [{

                    data: locValues,

                    backgroundColor: ['#0f172a', '#334155', '#475569', '#64748b', '#94a3b8', '#cbd5e1'], // Slate palette

                    borderWidth: 0

                }]

            }, commonOptions);


            // Render Chart 3: Deal Size (Bar/Trend equivalent)

            updateChart('dealSizeChart', 'bar', {

                labels: bucketLabels,

                datasets: [{

                    label: '# of Deals',

                    data: bucketValues,

                    backgroundColor: '#10b981', // Emerald 500

                    borderRadius: 4

                }]

            }, {

                ...commonOptions,

                plugins: { legend: { display: false } },

                scales: { y: { beginAtZero: true, ticks: { stepSize: 1 }, grid: { color: '#f1f5f9' } }, x: { grid: { display: false } } }

            });

        }


        function updateChart(canvasId, type, data, options) {

            const ctx = document.getElementById(canvasId).getContext('2d');

            

            if (charts[canvasId]) {

                charts[canvasId].destroy();

            }


            charts[canvasId] = new Chart(ctx, {

                type: type,

                data: data,

                options: options

            });

        }
